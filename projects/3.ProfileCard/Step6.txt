// using Card component inside of App component and passing name , title and bio with  props

export function Card({ name, title, bio }) {
  return (
    <div className="card">
      <h2>{name}</h2>
      <p className="card-title">{title}</p>
      <p>{bio}</p>
    </div>
  )
}

export function App() {
  return (
    <div className="flex-container">
      <Card name="Mark" title="Frontend developer" bio="I like to work with different frontend technologies and play video games." />
    </div>
  );
}
